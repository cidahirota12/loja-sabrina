⚙️ Configurar Sincronização com Google Sheets

⚠️ Atualize o código do Apps Script. Esta versão grava campos novos (estrutura de embalagem, nº da nota, chave). Se o script antigo continuar lá, esses campos se perdem a cada sincronização. Copie o código abaixo e substitua o antigo.
📖 Como configurar — passo a passo
Abra sua planilha Loja Sabrina — Fornecedores no Google Sheets (ou crie uma nova em sheets.google.com).
Clique em Extensões → Apps Script.
Apague TODO o código que estiver lá e cole este no lugar:
// Loja Sabrina — sincronizacao (versao 2) // Grava qualquer campo novo automaticamente, sem precisar mexer aqui de novo. const SHEET_MATRIZ = 'Matriz'; const SHEET_HISTORICO = 'Historico'; const COLUNAS_TEXTO = ['chaveNF','cnpj','nNF','notaId','cProd','ncm','cean','dataNF']; function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); return jsonResponse({ matriz: getSheetData(ss, SHEET_MATRIZ), historico: getSheetData(ss, SHEET_HISTORICO) }); } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (body.action === 'saveAll') { saveSheetData(ss, SHEET_MATRIZ, body.matriz); saveSheetData(ss, SHEET_HISTORICO, body.historico); return jsonResponse({ ok: true }); } if (body.action === 'saveMatriz') { saveSheetData(ss, SHEET_MATRIZ, body.data); return jsonResponse({ ok: true }); } return jsonResponse({ error: 'Acao invalida' }); } catch (err) { return jsonResponse({ error: err.message }); } } function getSheetData(ss, name) { const sheet = ss.getSheetByName(name); if (!sheet) return []; const vals = sheet.getDataRange().getValues(); if (vals.length < 2) return []; const headers = vals[0]; return vals.slice(1).map(function (row) { const obj = {}; headers.forEach(function (h, i) { if (h) obj[h] = row[i]; }); return obj; }); } function saveSheetData(ss, name, data) { let sheet = ss.getSheetByName(name); if (!sheet) sheet = ss.insertSheet(name); sheet.clear(); if (!data || !data.length) return; const headers = []; data.forEach(function (o) { Object.keys(o).forEach(function (k) { if (headers.indexOf(k) < 0) headers.push(k); }); }); headers.forEach(function (h, i) { if (COLUNAS_TEXTO.indexOf(h) >= 0) { sheet.getRange(1, i + 1, data.length + 1, 1).setNumberFormat('@'); } }); const rows = data.map(function (o) { return headers.map(function (h) { const v = o[h]; if (v === undefined || v === null) return ''; if (typeof v === 'object') return JSON.stringify(v); return v; }); }); sheet.getRange(1, 1, 1, headers.length).setValues([headers]); sheet.getRange(2, 1, rows.length, headers.length).setValues(rows); sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold'); sheet.setFrozenRows(1); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
